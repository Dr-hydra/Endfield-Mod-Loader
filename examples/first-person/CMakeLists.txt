cmake_minimum_required(VERSION 3.22)
project(EModLoaderFirstPerson VERSION 0.1.0 LANGUAGES CXX)
get_filename_component(EML_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(BETTER_ENDFIELD_SDK_INCLUDE "${EML_ROOT}/include" CACHE PATH "V1 SDK headers")
set(FIRST_PERSON_JSON_INCLUDE "${EML_ROOT}/native/shared/third_party" CACHE PATH "Directory containing nlohmann/json.hpp")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(first_person SHARED native/module.cpp)
target_include_directories(first_person PRIVATE "${BETTER_ENDFIELD_SDK_INCLUDE}" "${FIRST_PERSON_JSON_INCLUDE}")
set_target_properties(first_person PROPERTIES OUTPUT_NAME "example.first-person")
if(WIN32)
    target_compile_definitions(first_person PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(first_person PRIVATE /W4 /utf-8 /permissive-)
    set(PACKAGE_PLATFORM windows-x64)
elseif(ANDROID AND CMAKE_ANDROID_ARCH_ABI STREQUAL "arm64-v8a")
    target_link_libraries(first_person PRIVATE c++_static c++abi dl)
    target_compile_options(first_person PRIVATE -Wall -Wextra -fvisibility=hidden)
    target_link_options(first_person PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
    set(PACKAGE_PLATFORM android-arm64)
else()
    message(FATAL_ERROR "Supported package platforms: Windows x64 and Android arm64")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/module.json" MANIFEST)
if(WIN32)
    string(JSON MANIFEST REMOVE "${MANIFEST}" libraries android-arm64)
else()
    string(JSON MANIFEST REMOVE "${MANIFEST}" libraries windows-x64)
endif()
set(PACKAGE_DIR "${CMAKE_BINARY_DIR}/package")
file(MAKE_DIRECTORY "${PACKAGE_DIR}")
file(WRITE "${PACKAGE_DIR}/module.json" "${MANIFEST}\n")
file(COPY ui DESTINATION "${PACKAGE_DIR}")
set_target_properties(first_person PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PACKAGE_DIR}/native/${PACKAGE_PLATFORM}" LIBRARY_OUTPUT_DIRECTORY "${PACKAGE_DIR}/native/${PACKAGE_PLATFORM}")
foreach(CONFIG RELEASE DEBUG RELWITHDEBINFO MINSIZEREL)
    set_target_properties(first_person PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${CONFIG} "${PACKAGE_DIR}/native/${PACKAGE_PLATFORM}" LIBRARY_OUTPUT_DIRECTORY_${CONFIG} "${PACKAGE_DIR}/native/${PACKAGE_PLATFORM}")
endforeach()
if(WIN32)
    enable_testing()
    foreach(TEST first_person head_hide portable_mesh geometry profiles lifecycle)
        add_executable(fp_${TEST}_tests "tests/${TEST}_tests.cpp")
        target_include_directories(fp_${TEST}_tests PRIVATE "${BETTER_ENDFIELD_SDK_INCLUDE}" "${FIRST_PERSON_JSON_INCLUDE}")
        target_compile_definitions(fp_${TEST}_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
        target_compile_options(fp_${TEST}_tests PRIVATE /W4 /utf-8 /permissive-)
        add_test(NAME first_person_${TEST} COMMAND fp_${TEST}_tests)
    endforeach()
endif()
