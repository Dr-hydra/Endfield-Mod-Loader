cmake_minimum_required(VERSION 3.22)
project(EModLoaderAndroid LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(EML_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.." ABSOLUTE)
if(NOT EXISTS "${EML_DOBBY_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "Supply -PdobbySource=<Dobby source root>")
endif()
set(DOBBY_GENERATE_SHARED OFF CACHE BOOL "" FORCE)
set(DOBBY_DEBUG OFF CACHE BOOL "" FORCE)
add_subdirectory("${EML_DOBBY_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/dobby")
get_target_property(EML_DOBBY_SOURCES dobby SOURCES)
foreach(EML_DOBBY_SOURCE IN LISTS EML_DOBBY_SOURCES)
    if(EML_DOBBY_SOURCE MATCHES "(^|/)source/UserMode/ExecMemory/code-patch-tool-posix\\.cc$")
        list(REMOVE_ITEM EML_DOBBY_SOURCES "${EML_DOBBY_SOURCE}")
    endif()
endforeach()
set_property(TARGET dobby PROPERTY SOURCES "${EML_DOBBY_SOURCES}")
target_sources(dobby PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/core/code_patch_posix.cpp")
add_library(emodloader_host SHARED jni_bridge.cpp "${EML_ROOT}/src/native/runtime/module_runtime.cpp" "${EML_ROOT}/src/native/runtime/engine_resolver.cpp")
target_include_directories(emodloader_host PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${EML_ROOT}/include" "${EML_ROOT}/native/shared/third_party" "${EML_ROOT}/src/native" "${EML_DOBBY_SOURCE}/include")
target_link_libraries(emodloader_host PRIVATE dobby c++_static c++abi log dl)
target_link_options(emodloader_host PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
