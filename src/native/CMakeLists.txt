cmake_minimum_required(VERSION 3.25)
project(EModLoader VERSION 0.1.2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT WIN32)
    message(FATAL_ERROR "Use the Android CMake entry for arm64 builds.")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
get_filename_component(EML_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_subdirectory(vendor/minhook)
add_library(EMLRuntime STATIC runtime/module_runtime.cpp)
target_include_directories(EMLRuntime PUBLIC "${EML_ROOT}/include" "${EML_ROOT}/native/shared/third_party")
target_compile_definitions(EMLRuntime PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(EMLRuntime PRIVATE /W4 /utf-8 /permissive-)
target_link_libraries(EMLRuntime PUBLIC ws2_32)
add_executable(EModLoader.OfflineHost offline_main.cpp)
target_compile_options(EModLoader.OfflineHost PRIVATE /W4 /utf-8 /permissive-)
target_link_libraries(EModLoader.OfflineHost PRIVATE EMLRuntime minhook)
install(TARGETS EModLoader.OfflineHost RUNTIME DESTINATION runtime)
add_library(EModLoader.Host SHARED game_entry.cpp runtime/engine_resolver.cpp)
target_compile_options(EModLoader.Host PRIVATE /W4 /utf-8 /permissive-)
target_link_libraries(EModLoader.Host PRIVATE EMLRuntime minhook)
add_executable(EModLoader.Launcher launcher.cpp)
target_include_directories(EModLoader.Launcher PRIVATE "${EML_ROOT}/native/shared/third_party")
target_link_libraries(EModLoader.Launcher PRIVATE bcrypt)
target_compile_definitions(EModLoader.Launcher PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(EModLoader.Launcher PRIVATE /W4 /utf-8 /permissive-)
enable_language(ASM_MASM)
add_library(EModLoader.XInputProxy SHARED proxy/xinput_proxy.cpp proxy/xinput_relay.asm proxy/xinput.def)
target_include_directories(EModLoader.XInputProxy PRIVATE "${EML_ROOT}/native/shared/third_party")
target_compile_definitions(EModLoader.XInputProxy PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(EModLoader.XInputProxy PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/utf-8;/permissive->")
install(TARGETS EModLoader.XInputProxy RUNTIME DESTINATION runtime)
install(TARGETS EModLoader.Host EModLoader.Launcher RUNTIME DESTINATION runtime LIBRARY DESTINATION runtime)
enable_testing()
add_executable(EMLControlledTarget WIN32 "${EML_ROOT}/tests/native/controlled_target.cpp")
target_compile_definitions(EMLControlledTarget PRIVATE UNICODE _UNICODE)
add_library(EMLForeignXInput SHARED "${EML_ROOT}/tests/native/foreign_xinput.cpp")
add_executable(EMLHookTests "${EML_ROOT}/tests/native/hook_service_tests.cpp")
target_include_directories(EMLHookTests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(EMLHookTests PRIVATE EMLRuntime minhook)
target_compile_options(EMLHookTests PRIVATE /utf-8)
add_test(NAME product_hooks COMMAND EMLHookTests)
add_executable(EMLProxyControlledTarget WIN32 "${EML_ROOT}/tests/native/proxy_target.cpp")
target_compile_definitions(EMLProxyControlledTarget PRIVATE UNICODE _UNICODE)
add_executable(EMLProxyRelayTests "${EML_ROOT}/tests/native/xinput_relay_tests.cpp" proxy/xinput_relay.asm)
add_test(NAME xinput_relay COMMAND EMLProxyRelayTests)
